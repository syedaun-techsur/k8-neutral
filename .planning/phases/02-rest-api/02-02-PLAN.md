The `?q=` parameter is optional — if absent, return all notes. If present, filter by `title ILIKE '%q%'` (case-insensitive, PostgreSQL-native).

## Tasks

### Task 1: Create `app/api/notes/route.ts` — GET list + POST create

**File:** `app/api/notes/route.ts`

```ts
import { NextRequest, NextResponse } from 'next/server';
import { query, ready } from '@/lib/db';

export async function GET(request: NextRequest) {
  await ready;
  const q = request.nextUrl.searchParams.get('q');

  let result;
  if (q && q.trim() !== '') {
    result = await query(
      `SELECT id, title, body, pinned, created_at
         FROM notes
        WHERE title ILIKE $1
        ORDER BY pinned DESC, created_at DESC`,
      [`%${q}%`]
    );
  } else {
    result = await query(
      `SELECT id, title, body, pinned, created_at
         FROM notes
        ORDER BY pinned DESC, created_at DESC`
    );
  }

  return NextResponse.json(result.rows);
}

export async function POST(request: NextRequest) {
  await ready;
  const body = await request.json();
  const { title, body: noteBody = null, pinned = false } = body;

  if (!title || typeof title !== 'string' || title.trim() === '') {
    return NextResponse.json({ error: 'title is required' }, { status: 400 });
  }

  const result = await query(
    `INSERT INTO notes (title, body, pinned)
     VALUES ($1, $2, $3)
     RETURNING id, title, body, pinned, created_at`,
    [title.trim(), noteBody ?? null, Boolean(pinned)]
  );

  return NextResponse.json(result.rows[0], { status: 201 });
}
```

**Rules:**
- `await ready` before every DB operation (cheap no-op after first resolution)
- `request.nextUrl.searchParams.get('q')` — correct Next.js 14 way to read query params
- ILIKE with parameterized `%${q}%` — never string interpolation (SQL injection)
- Empty/whitespace-only `q` treats as "no filter" (return all)
- `title.trim() === ''` catches whitespace-only titles as invalid
- `RETURNING *` equivalent via explicit column list — returns the created note
- Status 201 on successful create
- Status 400 on missing/empty title
